<script>
async function loadPackages(country=""){
 let r = await fetch("backend/esim-api.php?action=packages");
 let d = await r.json();
 let c = document.getElementById("list");
 c.innerHTML="";
 d.data.forEach(p=>{
  c.innerHTML += `<div class="card">
   <h3>${p.title}</h3>
   <p>${p.data}GB - ${p.validity} days</p>
   <p class="price">$${p.price}</p>
   <button onclick="pay('${p.id}', ${p.price})">Pay & Buy</button>
  </div>`;
 });
}

function pay(id, price){
 const params = new URLSearchParams({amount:String(price), currency:'FC', desc:'Premium SpeedNet eSIM', bundleId:id, phone:(localStorage.getItem('sn_auth_user')||'')});
 window.location.href = 'payment.html?' + params.toString();
}


async function buy(id){
 let r = await fetch("backend/esim-api.php?action=buy",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({package_id:id})});
 let d = await r.json();
 localStorage.setItem("order",d.data.id);
fetch('admin/admin-orders.php',{method:'POST',body:JSON.stringify(d)});
 loadQR();
}

async function loadQR(){
 let id = localStorage.getItem("order");
 if(!id) return;
 let r = await fetch("backend/esim-api.php?action=order&id="+id);
 let d = await r.json();
 document.getElementById("qr").src = d.data.qr_code;
}

loadPackages(); loadQR();
