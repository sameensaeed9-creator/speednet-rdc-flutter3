const q = new URLSearchParams(location.search);
const amount = q.get('amount') || '0', currency = q.get('currency') || 'FC', desc = q.get('desc') || 'SpeedNet bundle', bundleId = q.get('bundleId') || '', phone = q.get('phone') || '';
document.getElementById('p-desc').textContent = desc; document.getElementById('p-amount').textContent = amount + ' ' + currency; document.getElementById('p-phone').textContent = phone || '-';
document.getElementById('f-amount').value=amount; document.getElementById('f-currency').value=currency; document.getElementById('f-desc').value=desc; document.getElementById('f-bundleId').value=bundleId; document.getElementById('f-phone').value=phone;
const mp=(window.SN_CONFIG && SN_CONFIG.manualPayment)||{};
document.getElementById('airtelNumber').textContent = '+243892399476'; document.getElementById('unitelNumber').textContent = '+243823778982'; document.getElementById('accountName1').textContent=mp.accountName||'SpeedNet RDC'; document.getElementById('accountName2').textContent=mp.accountName||'SpeedNet RDC';

const statusBox = document.getElementById('package-status-box');
const payBtn = document.querySelector('#manualPaymentForm button[type="submit"]');
async function liveCheckBeforePayment(){
  if(!bundleId){
    statusBox.style.display='block'; statusBox.textContent='Package ID missing. Please go back and select package again.';
    payBtn.disabled=true; return false;
  }
  statusBox.style.display='block'; statusBox.textContent='Checking latest package availability...';
  try{
    const r = await fetch('backend/package_status.php?fresh=1&package_id=' + encodeURIComponent(bundleId) + '&desc=' + encodeURIComponent(desc), {cache:'no-store'});
    const data = await r.json();
    const st = data.status || {};
    if(!st.available){
      statusBox.textContent = st.message || 'This package is currently unavailable. Please select another package.';
      payBtn.disabled = true;
      return false;
    }
    statusBox.textContent = 'Package available. You can submit payment proof.';
    payBtn.disabled = false;
    return true;
  }catch(err){
    statusBox.textContent='Live package check failed. Please refresh and try again.';
    payBtn.disabled=true;
    return false;
  }
}
liveCheckBeforePayment();
document.getElementById('manualPaymentForm').addEventListener('submit', async function(e){
  e.preventDefault();
  const ok = await liveCheckBeforePayment();
  if(!ok) return;
  const btn=this.querySelector('button'); btn.disabled=true; btn.textContent='Submitting...';
  try{const r=await fetch('backend/manual_payment_submit.php',{method:'POST',body:new FormData(this)}); const data=await r.json(); if(data.ok){location.href='success.html?manual_id='+encodeURIComponent(data.id);} else {alert(data.message||'Could not submit payment.'); btn.disabled=false; btn.textContent='Submit Payment for Approval';}}
  catch(err){alert('Upload failed. Please try again.'); btn.disabled=false; btn.textContent='Submit Payment for Approval';}
});
